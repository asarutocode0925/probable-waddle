    if not d then return end

    RL(d.T)

local function AH(p)
    if not S.Hitbox or p==L then return end

    if not c or HB[p.Name] then return end
    local t=c:FindFirstChild("UpperTorso") and {"Head","HumanoidRootPart","UpperTorso","LowerTorso"} or {"Head","HumanoidRootPart","Torso"}
    local sv={}
    for i=1,#t do
        local x=c:FindFirstChild(t[i])
        if x and x:IsA("BasePart") then
            sv[t[i]]={x.Size,x.Massless,x.CanCollide}
            x.Size=x.Size*S.HBV
            x.Massless=true
            x.CanCollide=false
        end
    end
    HB[p.Name]=sv
end

local function IV(pt,pl)
    if not pt then return false end
    local nw=tick()
    local c=VC[pl]
    if c and nw-c.t<.2 then return c.v end
    VFL[1]=C
    VFL[2]=L.Character
    RP.FilterDescendantsInstances=VFL
    local ry=RC(workspace,C.CFrame.Position,pt.Position-C.CFrame.Position,RP)
    local rs
    if not ry then rs=true
    else
        local m=ry.Instance:FindFirstAncestorOfClass("Model")
        rs=m and m==pt.Parent and m:FindFirstChildOfClass("Humanoid")~=nil
    end
    VC[pl]={t=nw,v=rs}
    return rs
end

local function PP(pt)
    if not S.AI or not pt then return pt.Position end
    local r=pt.Parent and pt.Parent:FindFirstChild("HumanoidRootPart")
    if not r then return pt.Position end
    local v=r.AssemblyLinearVelocity
    if S.PredictionType=="Acceleration" then
        local a=r:GetAttribute("Acceleration") or V3(0,0,0)
        return pt.Position+v*S.Pred+.5*a*S.Pred*S.Pred
    elseif S.PredictionType=="Gravity" then
        return pt.Position+v*S.Pred+V3(0,-workspace.Gravity*S.Pred*S.Pred*.5,0)
    end
    return pt.Position+v*S.Pred
end

local function GT()
    local myT=L.Team
    local tc=S.Team and myT
    local myC=L.Character
    local myP=myC and myC:FindFirstChild("HumanoidRootPart")
    if not myP then return nil end
    local mp=myP.Position
    local cs={}
    for _,p in GP(P) do
        if p==L then continue end
        if tc and p.Team==myT then continue end
        local ch=p.Character
        if not ch then continue end
        local h=ch:FindFirstChildOfClass("Humanoid")
        if not h or h.Health<=0 then continue end
        local pt=ch:FindFirstChild("Head") or ch:FindFirstChild("HumanoidRootPart")
        if not pt then continue end
        local d=(pt.Position-mp).Magnitude
        if d>S.MaxDist then continue end
        if S.VisCheck and not S.AimWallBang and not IV(pt,p) then continue end
        local sp,on=WV(C,PP(pt))
        if not (on and sp.Z>0) then continue end
        local sd=(V2(sp.X,sp.Y)-VP).Magnitude
        if sd>S.FOV then continue end
        table.insert(cs,{P=pt,D=d,SD=sd,H=h.Health,W=S.Whitelist[p.Name]==true})
    end
    if #cs==0 then return nil end
    table.sort(cs,function(a,b)
        if a.W~=b.W then return a.W end
        if S.AimPriority=="Health" then return a.H<b.H end
        if S.AimPriority=="Threat" then
            return (a.H*.5+a.D*.01)<(b.H*.5+b.D*.01)
        end
        return a.SD<b.SD
    end)
    local b=cs[1]
    if S.InstantSnap then
        pcall(function() C.CFrame=CF(C.CFrame.Position,PP(b.P)) end)
    end
    return b.P
end

local function EK(on)
    if AB.active and on then return end
    if not on and not AB.active then return end
    AB.active=on
    if on then
        if AB.orig.K==nil then AB.orig.K=L.Kick end
        if hookfunction and newcclosure then
            pcall(function()
                hookfunction(L.Kick,newcclosure(function(s,m)
                    if checkcaller and checkcaller() then return AB.orig.K(s,m) end
                    if s==L then return nil end
                    return AB.orig.K(s,m)
                end))
            end)
        end
        AB.conns.An=L.AncestryChanged:Connect(function(_,pr)
            if pr==nil then pcall(function() L.Parent=P end) end
        end)
        local function su(n)
            local l=n:lower()
            for _,w in {"kick","ban","detect","flag","punish","disconnect","exploit","cheat","terminate"} do
                if l:find(w,1,true) then return true end
            end
        end
        local function hR(o)
            if AB.hr[o] then return end
            AB.hr[o]=true
            if o.FireServer and hookfunction and newcclosure then
                pcall(function()
                    local og=o.FireServer
                    hookfunction(og,newcclosure(function(s,...)
                        local a={...}
                        local r=tostring(a[1] or ""):lower()
                        if r:find("kick",1,true) or r:find("ban",1,true) or r:find("detect",1,true) then
                            return nil
                        end
                        return og(s,...)
                    end))
                end)
            end
        end
        for _,o in RS:GetDescendants() do
            if (o:IsA("RemoteEvent") or o:IsA("RemoteFunction")) and su(o.Name) then hR(o) end
        end
        AB.conns.R=RS.DescendantAdded:Connect(function(o)
            if (o:IsA("RemoteEvent") or o:IsA("RemoteFunction")) and su(o.Name) then hR(o) end
        end)
        local cn=0
        AB.conns.W=R.Heartbeat:Connect(function()
            cn=cn+1
            if cn>=30 then
                cn=0
                if L.Parent~=P then pcall(function() L.Parent=P end) end
            end
        end)
    else
        if AB.orig.K and hookfunction then
            pcall(function() hookfunction(L.Kick,AB.orig.K) end)
        end
        for _,c in AB.conns do
            if c then pcall(function() c:Disconnect() end) end
        end
        AB.conns={}
        AB.hr={}
    end
end

local OLp=CFrame.Lerp
local function SCF(a,b,f)
    if S.AimSmoothing=="Exponential" then
        return OLp(a,b,f*f)
    elseif S.AimSmoothing=="Bezier" then
        local m=OLp(a,b,.5)
        local p1=OLp(a,m,f)
        local p2=OLp(m,b,f)
        return OLp(p1,p2,f)
    end
    return OLp(a,b,f)
end

        local v=mn+(mx-mn)*po
        if d==0 then v=F(v+.5)
        else v=F(v*10^d+.5)/10^d end
        local pc=(v-mn)/(mx-mn)
        fl_.Size=UD(pc,0,1,0)
        hd.Position=UD(pc,0,.5,0)
        vt.Text=string.format("%."..d.."f",v)
        cb(v)
    end
    ct.InputBegan:Connect(function(i)
        if i.UserInputType==Enum.UserInputType.Touch or i.UserInputType==Enum.UserInputType.MouseButton1 then
            local nw=tick()

                local po=(rvv-mn)/(mx-mn)

        CP(pn)
        RH(pn)
        task.wait(1)
        if ch and ch.Parent and ch:FindFirstChild("HumanoidRootPart") then
            MESP(p)
            AH(p)
        end
    end
    if p.Character then oc(p.Character) end
    p.CharacterAdded:Connect(oc)
    p.CharacterRemoving:Connect(function()
        CP(pn)
        RH(pn)
    end)
end
